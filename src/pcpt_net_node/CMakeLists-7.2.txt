cmake_minimum_required(VERSION 2.8.3)

project(pcpt_net_node)

add_compile_options(-std=c++11)
add_definitions(-Wall)      

SET(CMAKE_BUILD_TYPE release)

find_package(
    catkin REQUIRED COMPONENTS
    roscpp
	std_msgs
    pcl_ros
    comm_msg
    pcpt_net_lib
)

catkin_package(
  INCLUDE_DIRS include
  LIBRARIES 
  libpcpt_net_lib.so
  ${catkin_LIBRARIES}
)


include_directories(
    ${catkin_INCLUDE_DIRS}
    include
)

link_directories(
    ${CATKIN_DEVEL_PREFIX}/lib
    ${PCL_LIBRARY_DIRS}
)
find_package(CUDA REQUIRED QUIET)
include_directories(${CUDA_INCLUDE_DIRS})
if (CUDA_VERSION VERSION_LESS "11.0")
  set(CUDA_NVCC_FLAGS ${CUDA_NVCC_FLAGS} -std=c++11 -Wno-deprecated-gpu-targets)
else ()
  set(CUDA_NVCC_FLAGS ${CUDA_NVCC_FLAGS} -std=c++14 -Wno-deprecated-gpu-targets)
endif ()

set(CMAKE_INCLUDE_PATH /home/saite/TensorRT-7.2.1.6/include)
set(CMAKE_LIBRARY_PATH /home/saite/TensorRT-7.2.1.6/lib)

include_directories(${CMAKE_INCLUDE_PATH} )

set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS}  -D_MWAITXINTRIN_H_INCLUDED -D_FORCE_INLINES ")

if (CMAKE_VERSION VERSION_LESS "3.1")
    add_compile_options("-std=c++14 -fext-numeric-literals")
else ()
    set(CMAKE_CXX_STANDARD 14)
endif ()

file(GLOB SRC src/*.cc  src/*.cpp )

add_executable(pcpt_net_node ${SRC} )
target_link_libraries(pcpt_net_node
    libpcpt_net_lib.so
    ${catkin_LIBRARIES}
